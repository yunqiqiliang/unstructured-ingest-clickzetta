# =============================================================================
# ClickZetta Unstructured Ingest 环境配置示例
# =============================================================================
#
# 使用说明:
# 1. 复制此文件为 .env: cp .env.example .env
# 2. 填入您的实际配置信息
# 3. .env 文件已在 .gitignore 中，不会被提交到代码仓库
#

# =============================================================================
# ClickZetta 数据库配置 (必需)
# =============================================================================

# ClickZetta 服务地址
CLICKZETTA_SERVICE=your-clickzetta-service.com
# 或使用旧格式（兼容）
# cz_service=your-clickzetta-service.com

# ClickZetta 实例ID
CLICKZETTA_INSTANCE=your_instance_id
# 或使用旧格式（兼容）
# cz_instance=your_instance_id

# ClickZetta 工作空间
CLICKZETTA_WORKSPACE=your_workspace
# 或使用旧格式（兼容）
# cz_workspace=your_workspace

# ClickZetta 数据库Schema
CLICKZETTA_SCHEMA=your_schema
# 或使用旧格式（兼容）
# cz_schema=your_schema

# ClickZetta 用户名
CLICKZETTA_USERNAME=your_username
# 或使用旧格式（兼容）
# cz_username=your_username

# ClickZetta 密码
CLICKZETTA_PASSWORD=your_password
# 或使用旧格式（兼容）
# cz_password=your_password

# ClickZetta 虚拟集群
CLICKZETTA_VCLUSTER=your_vcluster
# 或使用旧格式（兼容）
# cz_vcluster=your_vcluster

# =============================================================================
# ClickZetta 高级配置 (可选)
# =============================================================================

# 连接超时时间（秒）
CLICKZETTA_CONNECTION_TIMEOUT=60

# 查询超时时间（秒）
CLICKZETTA_QUERY_TIMEOUT=1800

# =============================================================================
# AWS S3 配置 (可选 - 如需要S3存储)
# =============================================================================

# AWS 访问密钥ID
AWS_KEY=your_aws_access_key_id

# AWS 访问密钥Secret
AWS_SECRET=your_aws_secret_access_key

# S3存储桶名称
AWS_S3_NAME=s3://your-bucket-name/

# =============================================================================
# Unstructured API 配置 (可选 - 如需要文档处理)
# =============================================================================

# Unstructured API 密钥
UNSTRUCTURED_API_KEY=your_unstructured_api_key

# Unstructured API 端点
UNSTRUCTURED_URL=https://api.unstructuredapp.io/general/v0/general

# =============================================================================
# DashScope AI 模型配置 (可选 - 如需要AI功能)
# =============================================================================

# DashScope API 密钥
DASHSCOPE_API_KEY=your_dashscope_api_key

# =============================================================================
# 本地文件路径配置 (可选)
# =============================================================================

# 本地输入文件目录
LOCAL_FILE_INPUT_DIR=/path/to/your/input/files

# =============================================================================
# 配置说明
# =============================================================================
#
# ClickZetta 配置获取方式:
# 1. 登录 ClickZetta 控制台
# 2. 在"实例管理"中找到您的实例信息
# 3. 在"连接信息"中获取 service、instance、workspace 等参数
# 4. 使用您的账户凭据作为 username 和 password
#
# 环境变量优先级:
# 1. 标准格式 (CLICKZETTA_*) 优先
# 2. 兼容格式 (cz_*) 作为备选
#
# 测试环境变量:
# 运行 python -c "import os; print('✅ 配置正确' if os.getenv('CLICKZETTA_SERVICE') else '❌ 请检查配置')"
#
